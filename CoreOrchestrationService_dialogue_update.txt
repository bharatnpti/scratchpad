Key changes to CoreOrchestrationService.java to integrate with DialogueService:

1.  **Inject `DialogueService`**: Add `DialogueService` as a dependency in the constructor.

2.  **In `processIncomingText` method**:
    *   **Check for Active Clarification**: At the beginning, check if there's an active clarification for the current `conversationPlatformId` by calling `dialogueService.getDialogueState()`.
    *   **Handle Clarification Input**: If so, pass the `textInput` to `dialogueService.handleClarificationInput()`.
        *   If clarification is now complete (`dialogueState.isClarificationComplete()`):
            *   Retrieve the original intent and collected entities from the `dialogueState`.
            *   Re-attempt to process the original intent using the now complete set of entities.
            *   Clear the dialogue state using `dialogueService.clearDialogueState()`.
        *   If clarification is still ongoing:
            *   Formulate the next clarification question using `dialogueService.formulateClarificationQuestion()`.
            *   Return this question as the agent's response.
    *   **Normal Processing (No Active Clarification)**: If no active clarification, proceed as before with `nlpService.understandText()`.
    *   **Initiate Clarification (After NLP)**: After `nlpService.understandText()`, if the intent requires entities that are missing (this logic needs to be defined, e.g., a "CREATE_TASK" intent needs a 'description'):
        *   Identify missing entities.
        *   Call `dialogueService.initiateClarification()` with the intent, missing entities, and the `nlpResult`.
        *   Return the clarification question from `DialogueService` as the agent's response.
    *   **Response Formulation**: Instead of crafting response strings directly in `CoreOrchestrationService` for successful operations or errors, prepare a `ProcessingResult` (or similar DTO) and pass it to `dialogueService.formulateResponse()` to get the final user-facing message. This centralizes response generation.

    private List<String> checkForMissingEntities(String intent, Map<String, Object> entities) {
        List<String> missing = new ArrayList<>();
        switch (intent.toUpperCase()) {
            case "CREATE_TASK":
                if (!entities.containsKey("description") && !entities.containsKey("detail")) {
                    missing.add("description"); // Ask for description
                }
                // Add checks for other mandatory entities like 'dueDate' or 'assignee' if needed

                break;
            case "CALCULATOR":
                if (!entities.containsKey("operand1")) missing.add("operand1");
                if (!entities.containsKey("operand2")) missing.add("operand2");
                if (!entities.containsKey("operation")) missing.add("operation");
                break;
            // Add other intents and their mandatory entities
        }
        return missing;
    }

    private String routeIntent(StructuredNlpResult nlpResult, UserEntity user, ConversationEntity conversation, String originalUserInput) {
        String agentResponseText;
        // This part is similar to the original switch statement in processIncomingText
        // It would use dialogueService.formulateResponse with a ProcessingResult DTO
        // For brevity, direct string assignment is shown here.
        try {
            switch (nlpResult.intent().toUpperCase()) {
                case "CREATE_TASK":
                    String description = (String) nlpResult.entities().getOrDefault("description",
                        nlpResult.entities().getOrDefault("detail", "Task from: " + originalUserInput));
                    taskService.createTask(description, user, null, null, null, conversation, null);
                    agentResponseText = "Okay, I've created a task for you: " + description;
                    break;
                case "GET_WEATHER":
                case "CALCULATOR":
                case "ECHO":
                    String pluginName = nlpResult.intent().toLowerCase();
                    Map<String, Object> params = new HashMap<>(nlpResult.entities());
                    if ("calculator".equalsIgnoreCase(pluginName)) {
                         try {
                            Object op1 = params.get("operand1");
                            Object op2 = params.get("operand2");
                            if (op1 != null) params.put("operand1", Double.parseDouble(String.valueOf(op1))); else throw new NumberFormatException("operand1 missing");
                            if (op2 != null) params.put("operand2", Double.parseDouble(String.valueOf(op2))); else throw new NumberFormatException("operand2 missing");
                        } catch (NumberFormatException e) {
                            logger.warn("Could not parse calculator operands from entities: {}", nlpResult.entities());
                            // This case should ideally be caught by checkForMissingEntities and lead to clarification
                            throw new IllegalArgumentException("Calculator operands must be valid numbers. " + e.getMessage(), e);
                        }
                    }
                     if ("echo".equalsIgnoreCase(pluginName) && !params.containsKey("message")) {
                        params.put("message", originalUserInput); // If echo plugin needs a 'message' param
                    }
                    Map<String, Object> actionResult = actionExecutorService.executeAction(pluginName, params, null);
                    // Use DialogueService to format this properly
                    // agentResponseText = dialogueService.formulateResponse(new ProcessingResult(nlpResult.intent(), params, actionResult, "Action successful", false));
                    agentResponseText = "Action '" + pluginName + "' executed. Result: " + actionResult.toString();
                     if ("echo".equalsIgnoreCase(pluginName)) {
                         agentResponseText = actionResult.getOrDefault("echo_message", "Echo failed.").toString();
                    }
                    break;
                case "ERROR_NLP_PROCESSING":
                    agentResponseText = "I had trouble understanding your request due to an internal NLP processing error.";
                    logger.error("NLP processing error encountered: {}", nlpResult.entities().get("error"));
                    break;
                case "UNKNOWN_INTENT":
                    agentResponseText = "I'm not quite sure what you mean by that. Could you try rephrasing or being more specific?";
                    break;
                default:
                    logger.info("Unhandled intent '{}', attempting generic response for: {}", nlpResult.intent(), originalUserInput);
                    // List<NlpService.ChatMessageHistoryItem> chatHistory = getChatHistory(conversation.getConversationId(), null); // Pass null if current message ID is not relevant or available
                    // agentResponseText = nlpService.generateText(originalUserInput, chatHistory);
                    agentResponseText = "Thanks for your message. I'm still learning to handle intent: " + nlpResult.intent();
                    break;
            }
        } catch (IllegalArgumentException e) {
            logger.warn("Invalid argument during processing intent {}: {}", nlpResult.intent(), e.getMessage());
            agentResponseText = "There was an issue with the parameters for your request: " + e.getMessage();
        } catch (Exception e) {
            logger.error("Error processing intent {}: {}", nlpResult.intent(), e.getMessage(), e);
            agentResponseText = "I encountered an error trying to process your request. Please try again.";
        }
        return agentResponseText;
    }


    // Helper to get chat history (similar to previous version)
    private List<NlpService.ChatMessageHistoryItem> getChatHistory(UUID conversationId, UUID currentMessageIdToExclude) {
        List<MessageEntity> recentDbMessages = messageRepository.findByConversationConversationIdOrderByTimestampDesc(
            conversationId,
            PageRequest.of(0, 6, Sort.by("timestamp").descending())
        ).stream()
         .filter(m -> currentMessageIdToExclude == null || !m.getMessageId().equals(currentMessageIdToExclude))
         .limit(5)
         .collect(Collectors.toList());
        Collections.reverse(recentDbMessages);

        return recentDbMessages.stream()
            .map(msg -> new NlpService.ChatMessageHistoryItem(
                    msg.getRole().name(),
                    msg.getContent()))
            .collect(Collectors.toList());
    }

    private UserEntity getOrCreateUser(String platformId, String sourcePlatform) {
        return userRepository.findByPlatformIdAndSourcePlatform(platformId, sourcePlatform)
                .orElseGet(() -> {
                    UserEntity newUser = new UserEntity();
                    newUser.setPlatformId(platformId);
                    newUser.setSourcePlatform(sourcePlatform);
                    newUser.setDisplayName("User " + platformId);
                    logger.info("Creating new user for platformId: {} on platform: {}", platformId, sourcePlatform);
                    return userRepository.save(newUser);
                });
    }

    private ConversationEntity getOrCreateConversation(String platformId, String sourcePlatform, UserEntity user) {
        return conversationRepository.findByPlatformConversationIdAndSourcePlatform(platformId, sourcePlatform)
                .orElseGet(() -> {
                    ConversationEntity newConversation = new ConversationEntity();
                    newConversation.setPlatformConversationId(platformId);
                    newConversation.setSourcePlatform(sourcePlatform);
                    newConversation.setType("direct_message");
                    logger.info("Creating new conversation for platformId: {} on platform: {}", platformId, sourcePlatform);
                    return conversationRepository.save(newConversation);
                });
    }

    private UserEntity getAgentUser() {
        final String AGENT_PLATFORM_ID = "SYSTEM_AGENT";
        final String AGENT_SOURCE_PLATFORM = "INTERNAL";
        return userRepository.findByPlatformIdAndSourcePlatform(AGENT_PLATFORM_ID, AGENT_SOURCE_PLATFORM)
            .orElseGet(() -> {
                UserEntity agentUser = new UserEntity();
                agentUser.setPlatformId(AGENT_PLATFORM_ID);
                agentUser.setSourcePlatform(AGENT_SOURCE_PLATFORM);
                agentUser.setDisplayName("AI Agent");
                agentUser.setEmail("agent@system.internal");
                logger.info("Creating system agent user.");
                return userRepository.save(agentUser);
            });
    }
