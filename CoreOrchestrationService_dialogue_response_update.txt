package com.example.llmagentsystem.service;

import com.example.llmagentsystem.model.entity.ConversationEntity;
import com.example.llmagentsystem.model.entity.MessageEntity;
import com.example.llmagentsystem.model.entity.TaskEntity;
import com.example.llmagentsystem.model.entity.UserEntity;
import com.example.llmagentsystem.model.enums.UserRole;
import com.example.llmagentsystem.model.nlp.StructuredNlpResult;
import com.example.llmagentsystem.repository.ConversationRepository;
import com.example.llmagentsystem.repository.MessageRepository;
import com.example.llmagentsystem.repository.UserRepository;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Sort;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.OffsetDateTime;
import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.UUID;
import java.util.stream.Collectors;

@Service
public class CoreOrchestrationService {

    private static final Logger logger = LoggerFactory.getLogger(CoreOrchestrationService.class);

    private final NlpService nlpService;
    private final MemoryService memoryService;
    private final TaskService taskService;
    private final ActionExecutorService actionExecutorService;
    private final UserRepository userRepository;
    private final ConversationRepository conversationRepository;
    private final MessageRepository messageRepository;
    private final DialogueService dialogueService;

    @Autowired
    public CoreOrchestrationService(
            NlpService nlpService,
            MemoryService memoryService,
            TaskService taskService,
            ActionExecutorService actionExecutorService,
            UserRepository userRepository,
            ConversationRepository conversationRepository,
            MessageRepository messageRepository,
            DialogueService dialogueService) {
        this.nlpService = nlpService;
        this.memoryService = memoryService;
        this.taskService = taskService;
        this.actionExecutorService = actionExecutorService;
        this.userRepository = userRepository;
        this.conversationRepository = conversationRepository;
        this.messageRepository = messageRepository;
        this.dialogueService = dialogueService;
    }

    @Transactional
    public String processIncomingText(String textInput, String userPlatformId, String conversationPlatformId, String sourcePlatform) {
        logger.info("Processing text input: '{}' from user {} in conversation {} on platform {}",
                textInput, userPlatformId, conversationPlatformId, sourcePlatform);

        UserEntity user = getOrCreateUser(userPlatformId, sourcePlatform);
        ConversationEntity conversation = getOrCreateConversation(conversationPlatformId, sourcePlatform, user);
        conversation.setLastActivityTimestamp(OffsetDateTime.now());
        conversationRepository.save(conversation);

        MessageEntity userMessage = new MessageEntity();
        userMessage.setContent(textInput);
        userMessage.setConversation(conversation);
        userMessage.setUser(user);
        userMessage.setRole(UserRole.USER);
        userMessage.setTimestamp(OffsetDateTime.now());
        userMessage.setPlatformMessageId("simulated_platform_msg_id_" + UUID.randomUUID());
        MessageEntity savedUserMessage = messageRepository.save(userMessage);
        memoryService.storeMessageMemory(savedUserMessage);

        String agentResponseText;
        List<NlpService.ChatMessageHistoryItem> chatHistory = getChatHistory(conversation.getConversationId(), savedUserMessage.getMessageId());

        DialogueState dialogueState = dialogueService.getDialogueState(conversation.getPlatformConversationId());

        if (dialogueState != null && dialogueState.isAwaitingClarification()) {
            DialogueState updatedState = dialogueService.handleClarificationInput(
                conversation.getPlatformConversationId(), textInput, nlpService
            );

            if (updatedState.isClarificationComplete()) {
                logger.info("Clarification complete for conversation {}. Reprocessing original intent: {}",
                    conversation.getPlatformConversationId(), updatedState.getPendingIntent());

                StructuredNlpResult originalNlp = updatedState.getOriginalNlpResult();
                Map<String, Object> combinedEntities = new HashMap<>();
                if(originalNlp.entities() != null) combinedEntities.putAll(originalNlp.entities());
                combinedEntities.putAll(updatedState.getCollectedEntities());

                StructuredNlpResult reprocessedNlpResult = new StructuredNlpResult(
                    updatedState.getPendingIntent(),
                    combinedEntities,
                    originalNlp.rawResponse(),
                    originalNlp.confidence()
                );

                agentResponseText = routeIntentAndGetResponse(reprocessedNlpResult, user, conversation, textInput, chatHistory);
                dialogueService.clearDialogueState(conversation.getPlatformConversationId());
            } else {
                // Clarification still ongoing, formulate the next question
                agentResponseText = dialogueService.formulateClarificationQuestion(updatedState);
                 // Store this clarification question as agent's response
            }
        } else {
            StructuredNlpResult nlpResult = nlpService.understandText(textInput, chatHistory);
            logger.debug("Structured NLP Result - Intent: {}, Entities: {}", nlpResult.intent(), nlpResult.entities());

            List<String> missingEntities = checkForMissingEntities(nlpResult.intent(), nlpResult.entities());
            if (!missingEntities.isEmpty()) {
                String clarificationQuestion = dialogueService.initiateClarification(
                    conversation.getPlatformConversationId(), nlpResult.intent(), missingEntities, nlpResult
                );
                // Use ResponseGenerationContext for this question
                ResponseGenerationContext rgc = new ResponseGenerationContext(
                    ResponseReason.CLARIFICATION_NEEDED, nlpResult.intent(), clarificationQuestion, Map.of("missingEntities", missingEntities)
                );
                agentResponseText = dialogueService.formulateResponse(rgc, chatHistory);
            } else {
                agentResponseText = routeIntentAndGetResponse(nlpResult, user, conversation, textInput, chatHistory);
                dialogueService.clearDialogueState(conversation.getPlatformConversationId());
            }
        }

        MessageEntity agentMessage = new MessageEntity();
        agentMessage.setContent(agentResponseText);
        agentMessage.setConversation(conversation);
        agentMessage.setUser(getAgentUser());
        agentMessage.setRole(UserRole.AGENT);
        agentMessage.setTimestamp(OffsetDateTime.now());
        MessageEntity savedAgentMessage = messageRepository.save(agentMessage);
        memoryService.storeMessageMemory(savedAgentMessage);

        return agentResponseText;
    }

    private List<String> checkForMissingEntities(String intent, Map<String, Object> entities) {
        List<String> missing = new ArrayList<>();
        if (intent == null) return missing;

        switch (intent.toUpperCase()) {
            case "CREATE_TASK":
                if (entities == null || (!entities.containsKey("description") && !entities.containsKey("detail"))) {
                    missing.add("description");
                }
                break;
            case "CALCULATOR":
                 if (entities == null || !entities.containsKey("operand1")) missing.add("operand1");
                 if (entities == null || !entities.containsKey("operand2")) missing.add("operand2");
                 if (entities == null || !entities.containsKey("operation")) missing.add("operation");
                break;
        }
        return missing;
    }

    // Renamed from routeIntent to routeIntentAndGetResponse
    private String routeIntentAndGetResponse(StructuredNlpResult nlpResult, UserEntity user, ConversationEntity conversation, String originalUserInput, List<NlpService.ChatMessageHistoryItem> chatHistory) {
        ResponseGenerationContext rgc;
        String intent = nlpResult.intent() != null ? nlpResult.intent().toUpperCase() : "UNKNOWN_INTENT";

        try {
            switch (intent) {
                case "CREATE_TASK":
                    String description = (String) nlpResult.entities().getOrDefault("description",
                        nlpResult.entities().getOrDefault("detail", "Task from: " + originalUserInput));
                    TaskEntity createdTask = taskService.createTask(description, user, null, null, null, conversation, null);
                    rgc = new ResponseGenerationContext(ResponseReason.TASK_CREATED, intent, createdTask, nlpResult.entities());
                    break;
                case "GET_WEATHER": // Example, assuming a plugin exists
                case "CALCULATOR":
                case "ECHO":
                    String pluginName = intent.toLowerCase();
                    Map<String, Object> params = nlpResult.entities() != null ? new HashMap<>(nlpResult.entities()) : new HashMap<>();

                    if ("calculator".equalsIgnoreCase(pluginName)) {
                         try {
                            Object op1 = params.get("operand1");
                            Object op2 = params.get("operand2");
                            if (op1 != null) params.put("operand1", Double.parseDouble(String.valueOf(op1))); else throw new NumberFormatException("operand1 missing or null");
                            if (op2 != null) params.put("operand2", Double.parseDouble(String.valueOf(op2))); else throw new NumberFormatException("operand2 missing or null");
                        } catch (NumberFormatException e) {
                            logger.warn("Could not parse calculator operands from entities: {}", nlpResult.entities());
                            throw new IllegalArgumentException("Calculator operands must be valid numbers. " + e.getMessage(), e);
                        }
                    }
                     if ("echo".equalsIgnoreCase(pluginName) && (params == null || !params.containsKey("message")) ) {
                        params.put("message", originalUserInput);
                    }

                    Map<String, Object> actionResult = actionExecutorService.executeAction(pluginName, params, null);
                    rgc = new ResponseGenerationContext(ResponseReason.ACTION_EXECUTED, intent, actionResult, params);
                    break;
                case "ERROR_NLP_PROCESSING":
                    logger.error("NLP processing error encountered: {}", nlpResult.entities() != null ? nlpResult.entities().get("error") : "Unknown NLP error");
                    rgc = new ResponseGenerationContext(ResponseReason.ERROR_NLP, intent, nlpResult.entities().getOrDefault("error", "Unknown NLP error").toString(), nlpResult.entities());
                    break;
                case "UNKNOWN_INTENT":
                    rgc = new ResponseGenerationContext(ResponseReason.UNHANDLED_INTENT, intent, null, Map.of("userQuery", originalUserInput));
                    break;
                default: // Any other intent not specifically handled above
                    logger.info("Unhandled intent '{}', attempting generic response for: {}", intent, originalUserInput);
                    // This case might directly use nlpService.generateText or be wrapped in a GENERAL_NLP_RESULT
                    rgc = new ResponseGenerationContext(ResponseReason.UNHANDLED_INTENT, intent, nlpResult, Map.of("userQuery", originalUserInput));
                    break;
            }
        } catch (IllegalArgumentException e) {
            logger.warn("Invalid argument during processing intent {}: {}", intent, e.getMessage());
            rgc = new ResponseGenerationContext(ResponseReason.ERROR_INTERNAL, intent, e.getMessage(), nlpResult.entities());
        } catch (Exception e) {
            logger.error("Error processing intent {}: {}", intent, e.getMessage(), e);
            rgc = new ResponseGenerationContext(ResponseReason.ERROR_INTERNAL, intent, "An internal error occurred while processing your request.", nlpResult.entities());
        }
        return dialogueService.formulateResponse(rgc, chatHistory);
    }

    private List<NlpService.ChatMessageHistoryItem> getChatHistory(UUID conversationId, UUID currentMessageIdToExclude) {
        List<MessageEntity> recentDbMessages = messageRepository.findByConversationConversationIdOrderByTimestampDesc(
            conversationId,
            PageRequest.of(0, 6, Sort.by("timestamp").descending())
        ).stream()
         .filter(m -> currentMessageIdToExclude == null || !m.getMessageId().equals(currentMessageIdToExclude))
         .limit(5)
         .collect(Collectors.toList());
        Collections.reverse(recentDbMessages);

        return recentDbMessages.stream()
            .map(msg -> new NlpService.ChatMessageHistoryItem(
                    msg.getRole().name(),
                    msg.getContent()))
            .collect(Collectors.toList());
    }

    private UserEntity getOrCreateUser(String platformId, String sourcePlatform) {
        return userRepository.findByPlatformIdAndSourcePlatform(platformId, sourcePlatform)
                .orElseGet(() -> {
                    UserEntity newUser = new UserEntity();
                    newUser.setPlatformId(platformId);
                    newUser.setSourcePlatform(sourcePlatform);
                    newUser.setDisplayName("User " + platformId);
                    logger.info("Creating new user for platformId: {} on platform: {}", platformId, sourcePlatform);
                    return userRepository.save(newUser);
                });
    }

    private ConversationEntity getOrCreateConversation(String platformId, String sourcePlatform, UserEntity user) {
        return conversationRepository.findByPlatformConversationIdAndSourcePlatform(platformId, sourcePlatform)
                .orElseGet(() -> {
                    ConversationEntity newConversation = new ConversationEntity();
                    newConversation.setPlatformConversationId(platformId);
                    newConversation.setSourcePlatform(sourcePlatform);
                    newConversation.setType("direct_message");
                    logger.info("Creating new conversation for platformId: {} on platform: {}", platformId, sourcePlatform);
                    return conversationRepository.save(newConversation);
                });
    }

    private UserEntity getAgentUser() {
        final String AGENT_PLATFORM_ID = "SYSTEM_AGENT";
        final String AGENT_SOURCE_PLATFORM = "INTERNAL";
        return userRepository.findByPlatformIdAndSourcePlatform(AGENT_PLATFORM_ID, AGENT_SOURCE_PLATFORM)
            .orElseGet(() -> {
                UserEntity agentUser = new UserEntity();
                agentUser.setPlatformId(AGENT_PLATFORM_ID);
                agentUser.setSourcePlatform(AGENT_SOURCE_PLATFORM);
                agentUser.setDisplayName("AI Agent");
                agentUser.setEmail("agent@system.internal");
                logger.info("Creating system agent user.");
                return userRepository.save(agentUser);
            });
    }
}
